"""
WHAT: Tạo SparkSession với cấu hình MinIO/S3, shuffle, memory...
WHY: Gom vào 1 chỗ để các pipeline khác (products, reviews) dùng chung.
"""

from pyspark.sql import SparkSession
from .config import S3_ENDPOINT, S3_ACCESS_KEY, S3_SECRET_KEY

def create_spark_session(app_name: str = "ecommerce_dss_etl") -> SparkSession:
    """
    Create and return a SparkSession configured to talk to MinIO (S3 compatible).

    - Thiết lập fs.s3a.* để Spark đọc/ghi được MinIO.
    - Set một số option performance (shuffle partitions, driver/executor memory...).
    """
    spark = (
        SparkSession.builder.appName(app_name)
        .config("spark.hadoop.fs.s3a.endpoint", S3_ENDPOINT)
        .config("spark.hadoop.fs.s3a.access.key", S3_ACCESS_KEY)
        .config("spark.hadoop.fs.s3a.secret.key", S3_SECRET_KEY)
        .config("spark.hadoop.fs.s3a.path.style.access", "true")
        .config("spark.hadoop.fs.s3a.impl", "org.apache.hadoop.fs.s3a.S3AFileSystem")
        # Tùy chỉnh thêm từ file cũ nếu có
        .getOrCreate()
    )

    spark.sparkContext.setLogLevel("WARN")
    return spark

"""

Mỗi hàm tương ứng với 1 step trong sơ đồ "Data Standardization Tool".
"""

from pyspark.sql import DataFrame

    col, when, regexp_replace, trim, upper, to_timestamp, to_date, concat_ws, lit

# 1. DATA CLEANING
def clean_products(df_raw: DataFrame) -> DataFrame:
    """
    WHAT:
        Làm sạch các field cơ bản từ dữ liệu crawler:
        - đảm bảo có product_id, product_name, brand_name
        - convert giá về dạng số
        - tính data_quality_score

    WHY:
        Dữ liệu raw nhiều null, sai kiểu; nếu không clean
        thì mapping/standardize và ML/DWH sẽ bị sai.

    HOW:
        Áp dụng các rule y hệt trong clean_data() cũ, nhưng thêm comment rõ ràng.
    """
    df = df_raw

    # 1.1 Đảm bảo có product_id: nếu thiếu thì sinh từ URL
    df = df.withColumn(
        "product_id",
        when(
            col("product_id").isNull() | (trim(col("product_id")) == ""),
            # Fallback: hash từ URL (đảm bảo stable theo URL)
            # >>> Ở đây copy logic md5/url từ file cũ nếu bạn có
            col("product_id")  # TODO: thay bằng hash(url)
        ).otherwise(col("product_id"))
    )

    # 1.2 Chuẩn hóa product_name, null/blank -> "Unknown"
    df = df.withColumn(
        "product_name",
        when(
            col("product_name").isNull() | (trim(col("product_name")) == ""),
            lit("Unknown")
        ).otherwise(trim(col("product_name")))
    )

    # 1.3 Gộp brand & brand_name -> brand_name
    # (copy nguyên logic if/else từ clean_data() cũ)
    # - Ưu tiên brand_name non-null
    # - Nếu chỉ có brand -> cast sang brand_name
    # - Nếu không có gì -> "Unknown"
    df = df.withColumn(
        "brand_name",
        # TODO: dùng when/otherwise giống file cũ
        col("brand_name")
    )

    # 1.4 Chuẩn hóa giá: bỏ ký tự không phải số, cast long
    df = df.withColumn(
        "price_current",
        when(col("price_current").isNotNull(),
             regexp_replace(col("price_current").cast("string"), r"[^0-9]", ""))
        .otherwise(lit("0")).cast("long")
    )

    df = df.withColumn(
        "price_original",
        when(col("price_original").isNotNull(),
             regexp_replace(col("price_original").cast("string"), r"[^0-9]", ""))
        .otherwise(lit("0")).cast("long")
    )

    # 1.5 Chuẩn hóa discount_percent
    df = df.withColumn(
        "discount_percent",
        when(col("discount_percent").isNotNull(),
             regexp_replace(col("discount_percent").cast("string"), r"[^0-9.]", ""))
        .otherwise(lit("0")).cast("double")
    )

    # 1.6 Data quality score: record đủ tên & giá > 0
    df = df.withColumn(
        "data_quality_score",
        when(
            (col("product_name") != "Unknown") & (col("price_current") > 0),
            lit(1.0)
        ).otherwise(lit(0.0))
    )

    return df

# 3. DATA STANDARDIZATION
def standardize_products(df: DataFrame) -> DataFrame:
    """
    WHAT:
        Chuẩn hóa platform, brand, product_name, giá, thời gian snapshot.

    WHY:
        Tạo canonical form để:
        - hash ID chuẩn
        - join dim/fact ổn định

    HOW:
        - source_platform_std: tiki/tiki_mass_crawl -> 'tiki', lazada/... -> 'lazada'
        - brand_std: UPPER(TRIM(brand_name))
        - product_name_std: trim + gộp space, null -> 'Unknown'
        - price_current_vnd, price_original_vnd: cast double
        - crawl_ts: to_timestamp(crawl_date), snapshot_date: to_date(crawl_ts)
    """
    df_std = df

    # Platform chuẩn
    df_std = df_std.withColumn("platform_raw", trim(col("source_platform")).cast("string"))
    df_std = df_std.withColumn(
        "source_platform_std",
        when(col("platform_raw").isin("tiki", "tiki_mass_crawl"), lit("tiki"))
        .when(col("platform_raw").isin("lazada", "lazada_mass_crawl"), lit("lazada"))
        .otherwise(col("platform_raw"))
    )

    # Brand chuẩn
    df_std = df_std.withColumn("brand_std", upper(trim(col("brand_name"))))

    # Product name chuẩn
    df_std = df_std.withColumn(
        "product_name_std",
        when(
            col("product_name").isNotNull() & (trim(col("product_name")) != ""),
            regexp_replace(trim(col("product_name")), r"\s+", " ")
        ).otherwise(lit("Unknown"))
    )

    # Giá dạng double
    df_std = df_std.withColumn("price_current_vnd", col("price_current").cast("double"))
    df_std = df_std.withColumn("price_original_vnd", col("price_original").cast("double"))

    # Thời gian snapshot
    df_std = df_std.withColumn(
        "crawl_ts",
        # TODO: copy logic parse timestamp chính xác từ file cũ (2 format ISO & date-only)
        to_timestamp(col("crawl_date"))
    )
    df_std = df_std.withColumn("snapshot_date", to_date(col("crawl_ts")))

    return df_std

# 4. SYNCHRONIZE IDENTIFIERS
def sync_product_identifiers(df: DataFrame) -> DataFrame:
    """
    WHAT:
        Sinh & đồng bộ các ID chuẩn:
        - product_id_std
        - global_product_id_synced
        - product_master_id
        - sku_id

    WHY:
        Đảm bảo mỗi sản phẩm & SKU có khóa ổn định để:
        - dedup
        - join dim/fact
        - phân tích cross-platform

    HOW:
        - product_id_std = trim(product_id)
        - global_product_id_synced = source_platform_std + '_' + product_id_std
        - product_master_id = sha2(brand_std || product_name_std || category_std)
        - sku_id = sha2(source_platform_std || seller_name || product_id_std)
    """
    from pyspark.sql.functions import sha2, concat_ws

    df_id = df
    df_id = df_id.withColumn("product_id_std", trim(col("product_id")).cast("string"))

    df_id = df_id.withColumn(
        "global_product_id_synced",
        when(
            col("product_id_std").isNotNull() & (col("product_id_std") != ""),
            concat_ws("_", col("source_platform_std"), col("product_id_std"))
        ).otherwise(trim(col("global_product_id")))
    )

    df_id = df_id.withColumn(
        "product_master_id",
        sha2(concat_ws("||", col("brand_std"), col("product_name_std"), col("category_std")), 256)
    )

    df_id = df_id.withColumn(
        "sku_id",
        sha2(concat_ws("||", col("source_platform_std"), col("seller_name"), col("product_id_std")), 256)
    )

    return df_id

# 5. DEDUPLICATE
def deduplicate_products(df: DataFrame) -> DataFrame:
    """
    WHAT:
        Bỏ duplicate theo (global_product_id_synced, snapshot_date).

    WHY:
        Đảm bảo grain = 1 sản phẩm / platform / ngày.

    HOW:
        - Nếu có snapshot_date: dropDuplicates(["global_product_id_synced", "snapshot_date"])
        - Nếu không: cảnh báo & dropDuplicates theo global_product_id_synced
    """
    if "snapshot_date" in df.columns:
        return df.dropDuplicates(["global_product_id_synced", "snapshot_date"])
    return df.dropDuplicates(["global_product_id_synced"])

# 6. VALIDATE

# app/etl/pipeline_main.py

from .spark_session import create_spark_session
from .extract import load_raw_products, load_raw_reviews
from .product_transforms import (
    clean_products, map_product_categories, standardize_products,
    sync_product_identifiers, deduplicate_products, validate_products,
)
from .product_aggregation import aggregate_products_daily, save_cleaned_products
from .dwh_loader import load_dimensions, load_fact_product_daily
from .review_transforms import run_review_transform_pipeline  # bạn tự wrap các step review vào 1 hàm
from .review_loader import load_fact_review_star, load_fact_review_daily_star
from .config import PROCESS_REVIEWS

def run_etl():
    """
    Orchestrate toàn bộ ETL:

    1. Extract raw từ MinIO
    2. Transform product (clean -> map category -> standardize -> sync ID -> dedup -> validate)
    3. Aggregate product daily & lưu cleaned
    4. Load dim_* & fact_product_daily vào DWH
    5. (Optional) chạy pipeline review & load fact_review*
    """
    spark = create_spark_session()
    try:
        # 1. EXTRACT
        df_raw_products = load_raw_products(spark)

        # 2. TRANSFORM PRODUCTS
        df_clean = clean_products(df_raw_products)
        df_mapped = map_product_categories(df_clean)
        df_std = standardize_products(df_mapped)
        df_ids = sync_product_identifiers(df_std)
        df_dedup = deduplicate_products(df_ids)
        validate_products(df_dedup)

        # 3. AGGREGATE & SAVE
        df_daily = aggregate_products_daily(df_dedup)
        save_cleaned_products(df_dedup, spark)

        # 4. LOAD DWH (DIM + FACT PRODUCTS)
        conn, mappings = load_dimensions(df_dedup)
        load_fact_product_daily(df_daily, conn, mappings)

        # 5. REVIEWS (optional)
        if PROCESS_REVIEWS:
            df_reviews_dedup, df_reviews_daily = run_review_transform_pipeline(
                spark, mappings
            )
            load_fact_review_star(df_reviews_dedup, conn, mappings)
            load_fact_review_daily_star(df_reviews_daily, conn, mappings)
    finally:
        spark.stop()

if __name__ == "__main__":
    run_etl()
