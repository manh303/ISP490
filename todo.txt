
from enum import Enum
from datetime import date
from typing import List, Optional
from pydantic import BaseModel, Field, root_validator  # hoặc model_validator nếu pydantic v2


class PriceScopeMode(str, Enum):
    BY_CATEGORY = "by_category"
    BY_PRODUCT = "by_product"


class PricePredictionRequest(BaseModel):
    from_date: date
    to_date: date

    # Lọc theo platform/category (dùng cho BY_CATEGORY)
    platforms: List[str] = Field(default_factory=list)
    categories: List[str] = Field(default_factory=list)

    # Lọc theo product cụ thể (dùng cho BY_PRODUCT)
    product_keys: List[str] = Field(default_factory=list)

    page: int = 1
    page_size: int = 50

    # Chế độ
    scope_mode: PriceScopeMode = PriceScopeMode.BY_CATEGORY
    top_n: int = 50  # chỉ có ý nghĩa khi BY_CATEGORY

    max_discount_pct: float = 0.15
    min_margin_pct: float = 0.10
    min_confidence: float = 0.70
    min_price_change_pct: float = 0.02

    @root_validator
    def validate_scope(cls, values):
        scope_mode = values.get("scope_mode")
        categories = values.get("categories") or []
        product_keys = values.get("product_keys") or []

        if scope_mode == PriceScopeMode.BY_CATEGORY:
            if not categories:
                raise ValueError("categories must not be empty when scope_mode='by_category'")
        elif scope_mode == PriceScopeMode.BY_PRODUCT:
            if not product_keys:
                raise ValueError("product_keys must not be empty when scope_mode='by_product'")
        return values
Nếu bạn đang dùng Pydantic v2 thì đổi @root_validator thành @model_validator(mode="after"), logic tương tự.

async def run_price_prediction_dss(self, payload: Dict[str, Any]) -> Dict[str, Any]:
    scope_mode = payload.get("scope_mode", "by_category")
    from_date = payload["from_date"]
    to_date = payload["to_date"]
    platforms = payload.get("platforms", [])
    categories = payload.get("categories", [])
    product_keys = payload.get("product_keys", [])
    top_n = payload.get("top_n", 50)

    max_discount_pct = payload.get("max_discount_pct", 0.15)
    min_margin_pct = payload.get("min_margin_pct", 0.10)
    min_confidence = payload.get("min_confidence", 0.70)
    min_price_change_pct = payload.get("min_price_change_pct", 0.02)

    if scope_mode == "by_category":
        result_rows = await self._run_price_by_category(
            from_date, to_date,
            platforms, categories,
            top_n,
            max_discount_pct, min_margin_pct,
            min_confidence, min_price_change_pct,
        )
    elif scope_mode == "by_product":
        result_rows = await self._run_price_by_product_keys(
            from_date, to_date,
            product_keys,
            max_discount_pct, min_margin_pct,
            min_confidence, min_price_change_pct,
        )
    else:
        raise ValueError(f"Unsupported scope_mode: {scope_mode}")

    # Ở dưới bạn build KPIs, summary, actions từ result_rows
    summary = self._build_price_dss_summary(result_rows, scope_mode)
    return summary
3.2. Hàm con cho BY_CATEGORY
Pseudo-SQL: (bạn map lại cho đúng tên bảng/column thực tế)

python
Sao chép mã
async def _run_price_by_category(
    self, from_date, to_date,
    platforms, categories,
    top_n,
    max_discount_pct, min_margin_pct,
    min_confidence, min_price_change_pct,
):
    sql = """
    WITH candidate AS (
        SELECT
            p.product_key,
            p.product_name,
            pl.platform_code,
            c.category_key,
            c.category_name,
            d.date_value,
            f.current_price,
            ml.predicted_price,
            ml.confidence,
            -- ví dụ tính % chênh
            (ml.predicted_price - f.current_price) / NULLIF(f.current_price, 0) AS price_change_pct,
            -- revenue giả lập
            (ml.predicted_price * f.sales_qty) AS predicted_revenue,
            (f.current_price * f.sales_qty) AS current_revenue,
            (ml.predicted_price * f.sales_qty) - (f.current_price * f.sales_qty) AS revenue_uplift
        FROM dwh.fact_product_daily f
        JOIN dwh.dim_product p   ON f.product_key = p.product_key
        JOIN dwh.dim_platform pl ON p.platform_key = pl.platform_key
        JOIN dwh.dim_category c  ON p.category_key = c.category_key
        JOIN dwh.dim_date d      ON f.date_key = d.date_key
        JOIN ml.fact_price_prediction ml
          ON ml.product_key = f.product_key
         AND ml.date_key = f.date_key
        WHERE d.date_value BETWEEN $1 AND $2
          AND ($3::text[] IS NULL OR pl.platform_code = ANY($3))
          AND ($4::text[] IS NULL OR c.category_key = ANY($4))
          AND ml.confidence >= $5
          AND ABS((ml.predicted_price - f.current_price) / NULLIF(f.current_price, 0)) >= $6
          -- TODO: filter thêm max_discount_pct, min_margin_pct
    )
    SELECT *
    FROM candidate
    ORDER BY revenue_uplift DESC
    LIMIT $7;
    """
    rows = await self.conn.fetch(
        sql,
        from_date, to_date,
        platforms or None,
        categories or None,
        min_confidence,
        min_price_change_pct,
        top_n,
    )
    return rows
Ý chính: lọc theo platform + category + rule ML → sort theo revenue_uplift DESC → lấy top_n.
Sau đó bạn build response (list sản phẩm, gợi ý giá, uplift…).
