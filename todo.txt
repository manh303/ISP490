    async def _resolve_model(
        self,
        model_type: str,
        model_name: Optional[str],
        model_version: Optional[str],
    ) -> MLModelResponse:
        """
        Chọn model từ ml.dim_ml_model theo logic:
        - Nếu truyền model_name/model_version → dùng exact match
        - Nếu không → lấy model mới nhất có status='active' cho model_type
        
        Raises RuntimeError nếu không tìm thấy model phù hợp.
        """
        query = """
            SELECT model_sk, model_name, model_type, model_version,
                   training_data_until, metrics, status, created_at
            FROM ml.dim_ml_model
            WHERE model_type = $1
              AND status = 'active'
        """
        params = [model_type]

        if model_name:
            query += f" AND model_name = ${len(params)+1}"
            params.append(model_name)
        if model_version:
            query += f" AND model_version = ${len(params)+1}"
            params.append(model_version)

        query += " ORDER BY created_at DESC LIMIT 1"

        row = await self.db.fetchrow(query, *params)
        if not row:
            raise RuntimeError(f"No active model found for type={model_type}")

        metrics = row["metrics"]
        if isinstance(metrics, str):
            metrics = json.loads(metrics)

        return MLModelResponse(
            model_sk=row["model_sk"],
            model_name=row["model_name"],
            model_type=row["model_type"],
            model_version=row["model_version"],
            training_data_until=row["training_data_until"],
            metrics=metrics,
            status=row["status"],
            created_at=row["created_at"],
        )

BENEFIT: 
- Tự động chọn model mới nhất nếu FE không chỉ định
- Support A/B testing: FE có thể chọn specific version
- Centralized error handling khi không có model


3.2. STEP 2: Tạo module ml_runtime.py để load & cache models
--------------------------------------------------------------------------------
File: backend/app/ml_runtime.py (NEW FILE)

import os
import pickle
from functools import lru_cache
from pathlib import Path
from typing import Tuple, Any

# Đọc từ env variable, mặc định là thư mục "models" trong root
MODEL_DIR = Path(os.getenv("ML_MODEL_DIR", "models"))

@lru_cache(maxsize=10)
def load_price_model(model_name: str, model_version: str) -> Any:
    """
    Load price prediction model từ file .pkl
    Quy ước tên file: {model_name}_{model_version}.pkl
    VD: price_forecast_rf_v1.0.pkl
    
    Cache 10 models gần nhất để tránh reload
    """
    path = MODEL_DIR / f"{model_name}_{model_version}.pkl"
    if not path.exists():
        raise FileNotFoundError(f"Model file not found: {path}")
    
    with open(path, "rb") as f:
        model = pickle.load(f)
    return model

@lru_cache(maxsize=10)
def load_sentiment_pipeline(model_name: str, model_version: str) -> Tuple[Any, Any]:
    """
    Load sentiment model bundle từ file .pkl
    Bundle format: {"vectorizer": TfidfVectorizer, "model": LogisticRegression}
    
    Returns: (vectorizer, classifier)
    """
    path = MODEL_DIR / f"{model_name}_{model_version}.pkl"
    if not path.exists():
        raise FileNotFoundError(f"Model file not found: {path}")
    
    with open(path, "rb") as f:
        bundle = pickle.load(f)
    
    # Validate bundle structure
    if not isinstance(bundle, dict) or "vectorizer" not in bundle or "model" not in bundle:
        raise ValueError(f"Invalid model bundle format in {path}")
    
    return bundle["vectorizer"], bundle["model"]

NOTES:
- @lru_cache giúp model được load một lần, reuse cho các requests sau
- maxsize=10: cache tối đa 10 models (đủ cho dev/staging)
- Production: có thể tăng maxsize hoặc dùng Redis cache
- Error handling: raise FileNotFoundError rõ ràng nếu file không tồn tại

    async def online_price_prediction(
        self, payload: OnlinePricePredictionRequest
    ) -> OnlinePricePredictionResponse:
        """
        Online price prediction sử dụng model thật từ registry + file.
        
        Flow:
        1. Resolve model từ registry
        2. Lấy features (từ payload hoặc query DWH)
        3. Load model & predict
        4. Return kết quả + latency
        """
        import time
        import numpy as np
        from app.ml_runtime import load_price_model
        
        t0 = time.perf_counter()

        # 1) Resolve model từ registry
        model_info = await self._resolve_model(
            model_type="price",
            model_name=payload.model_name,
            model_version=payload.model_version,
        )

        # 2) Chuẩn hoá features
        # Ưu tiên lấy từ payload, nếu thiếu thì query DWH
        current_price = payload.current_price
        avg_rating = payload.avg_rating
        review_count = payload.review_count

        if current_price is None or avg_rating is None or review_count is None:
            # Query snapshot mới nhất từ fact_product_daily
            snap_sql = """
                SELECT fpd.min_price, fpd.max_price,
                       fpd.total_review_count, fpd.avg_rating
                FROM dwh.fact_product_daily fpd
                JOIN dwh.dim_product dp ON dp.product_sk = fpd.product_sk
                JOIN dwh.dim_platform pl ON pl.platform_sk = fpd.platform_sk
                JOIN dwh.dim_date d ON d.date_sk = fpd.date_sk
                WHERE dp.product_key = $1
                  AND pl.platform_code = $2
                ORDER BY d.date_value DESC
                LIMIT 1
            """
            snap = await self.db.fetchrow(
                snap_sql,
                payload.product_key,
                payload.platform_code,
            )
            if not snap:
                raise ValueError(f"No snapshot found for product {payload.product_key}")

            min_price = float(snap["min_price"])
            max_price = float(snap["max_price"])
            total_review_count = int(snap["total_review_count"] or 0)
            avg_rating = float(snap["avg_rating"] or 0.0)
        else:
            # Dùng current_price cho cả min/max (simplified)
            min_price = max_price = float(current_price)
            total_review_count = int(review_count or 0)
            avg_rating = float(avg_rating or 0.0)

        # Tạo feature vector (tuỳ model, có thể cần thêm features)
        X = np.array([[min_price, max_price, total_review_count, avg_rating]])

        # 3) Load model & predict
        model = load_price_model(model_info.model_name, model_info.model_version)
        pred = float(model.predict(X)[0])

        # Optional: tính CI nếu model support (quantile regression, etc)
        ci_lower = None
        ci_upper = None
        # TODO: implement CI calculation if needed

        latency_ms = int((time.perf_counter() - t0) * 1000)

        return OnlinePricePredictionResponse(
            predicted_price=pred,
            ci_lower=ci_lower,
            ci_upper=ci_upper,
            model_name=model_info.model_name,
            model_version=model_info.model_version,
            latency_ms=latency_ms,

    async def online_sentiment(
        self,
        payload: OnlineSentimentRequest,
    ) -> OnlineSentimentResponse:
        """
        Online sentiment analysis sử dụng model thật.
        
        Flow:
        1. Resolve model từ registry
        2. Load vectorizer + classifier
        3. Transform text → predict_proba
        4. Return label, score, latency
        """
        import time
        import numpy as np
        from app.ml_runtime import load_sentiment_pipeline
        
        t0 = time.perf_counter()

        # 1) Resolve model từ registry
        model_info = await self._resolve_model(
            model_type="sentiment",
            model_name=payload.model_name,
            model_version=payload.model_version,
        )

        # 2) Load pipeline (vectorizer + classifier)
        vectorizer, clf = load_sentiment_pipeline(
            model_info.model_name,
            model_info.model_version,
        )

        # 3) Transform & predict
        X = vectorizer.transform([payload.review_text])
        proba = clf.predict_proba(X)[0]
        
        # Lấy label có xác suất cao nhất
        idx = int(np.argmax(proba))
        label = str(clf.classes_[idx])
        score = float(proba[idx])

        latency_ms = int((time.perf_counter() - t0) * 1000)

        return OnlineSentimentResponse(
            label=label,
            score=score,
            model_name=model_info.model_name,
            model_version=model_info.model_version,
            latency_ms=latency_ms,
        )
